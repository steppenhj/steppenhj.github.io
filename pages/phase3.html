Key Improvement: DMA & Queue

stm32_uart.c (Interrupt to Queue)
// UART Rx Complete Callback (ISR Context)
void HAL_UARTEx_RxEventCallback(...) {
    BaseType_t xHigherPriorityTaskWoken = pdFALSE;
    
    // 1. Send Data to Queue directly from ISR
    xQueueSendFromISR(CommQueue, &rx_data, &xHigherPriorityTaskWoken);
    
    // 2. Restart DMA
    HAL_UARTEx_ReceiveToIdle_DMA(&huart2, rx_buffer, 64);
    
    // 3. Context Switch if needed
    portYIELD_FROM_ISR(xHigherPriorityTaskWoken);
}
